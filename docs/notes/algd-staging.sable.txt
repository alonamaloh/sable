// Single-limb constructor.
/// post #[label(value)] natVal result.limbs = d
/// post #[label(len)]   result.limbs.len ≤ 1
fn nat1(u32 d) -> Nat {
    mut [u32] xs = alloc_array<u32>(1, 0);
    xs[0] = d;
    if (d == 0) {
        return Nat::from_prefix(&xs, 0);
    }
    return Nat::from_prefix(&xs, 1);
}

// R' = B·R + d: shift the limbs up one position and drop `d` in at the
// bottom.
/// pre  r.limbs.len ≤ 4611686018427387904
/// post #[label(value)] natVal result.limbs
///          = 4294967296 * natVal r.limbs + d
/// post #[label(len)]   result.limbs.len ≤ r.limbs.len + 1
fn shift_in(&Nat r, u32 d) -> Nat {
    u64 rl = r.limbs.len;
    if (rl == 0) {
        return nat1(d);
    }
    mut [u32] xs = alloc_array<u32>(rl + 1, 0);
    xs[0] = d;
    /// invariant ∀ k, 0 ≤ k → k < i → xs.get (k + 1) = r.limbs.get k
    /// invariant xs.get 0 = d
    for (u64 i : range(rl)) {
        xs[i + 1] = r.limbs[i];
    }
    return Nat::from_prefix(&xs, rl + 1);
}

// Normalization factor: the doubling loop with the relaxed guard
// (`t + m ≤ 2^31`), giving quarter-normalization — see the deviation
// note in docs/notes/algd-probe.lean.
/// pre  1 ≤ t0
/// pre  t0 ≤ 4294967295
/// post #[label(lo)]  1073741824 < t0 * result
/// post #[label(hi)]  t0 * result + result ≤ 4294967296
/// post #[label(pos)] 1 ≤ result
/// post #[label(cap)] result ≤ 2147483648
fn norm_factor(u64 t0) -> u64 {
    mut u64 t = t0;
    mut u64 m = 1;
    /// invariant t = t0 * m
    /// invariant 1 ≤ m
    /// invariant t + m ≤ 4294967296
    /// variant   4294967296 - t
    while (t + m <= 2147483648) {
        t = t + t;
        m = m + m;
    }
    return m;
}

/// pre  natVal b.limbs ≥ 1
/// pre  a.limbs.len + b.limbs.len ≤ 2305843009213693952
/// post #[label(quot)] natVal result.limbs = natVal a.limbs / natVal b.limbs
fn div_fast(&Nat a, &Nat b) -> Nat {
    u64 n = b.limbs.len;
    /// assert #[label(n_pos)] 1 ≤ n
    u64 mfac = norm_factor(widen<u64>(b.limbs[n - 1]));
    var mn = nat1(narrow<u32>(mfac));
    /// assert #[label(mn_val)] natVal mn.limbs = mfac
    var vn = b * mn;
    /// assert #[label(vn_val)]  natVal vn.limbs = natVal b.limbs * mfac
    /// assert #[label(vn_pos)]  1 ≤ natVal vn.limbs
    /// assert #[label(vn_len)]  vn.limbs.len = n
    /// assert #[label(vn_top)]  1073741824 < vn.limbs.get (n - 1)
    var u = a * mn;
    /// assert #[label(u_val)]   natVal u.limbs = natVal a.limbs * mfac
    /// assert #[label(u_len)]   u.limbs.len ≤ a.limbs.len + 1
    var mut q = nat1(0);
    var mut r = nat1(0);
    /// assert #[label(q0)] natVal q.limbs = 0
    /// assert #[label(r0)] natVal r.limbs = 0
    mut u64 j = u.limbs.len;
    /// invariant j ≤ u.limbs.len
    /// invariant natVal q.limbs * natVal vn.limbs + natVal r.limbs
    ///             = valIn u.limbs j u.limbs.len
    /// invariant natVal r.limbs < natVal vn.limbs
    /// invariant q.limbs.len ≤ u.limbs.len - j + 1
    /// variant   j
    while (j > 0) {
        j = j - 1;
        var r2 = shift_in(&r, u.limbs[j]);
        /// assert #[label(r2_val)] natVal r2.limbs
        ///           = 4294967296 * natVal r.limbs + u.limbs.get j
        /// assert #[label(r2_bound)] natVal r2.limbs
        ///           < 4294967296 * natVal vn.limbs
